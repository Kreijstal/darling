function(add_framework name)
 cmake_parse_arguments(F "" "TARGET_NAME" "" ${ARGN})
 add_library(${F_TARGET_NAME} SHARED "${CMAKE_CURRENT_SOURCE_DIR}/empty.c")
 set_property(TARGET ${F_TARGET_NAME} PROPERTY DYLIB_INSTALL_NAME "/System/Library/Frameworks/${name}.framework/Versions/A/${name}")
endfunction()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../src/frameworks/load-only" fallback)
foreach(name JavaScriptCore HealthKit)
 if(TARGET load_only_${name})
  message(FATAL_ERROR "fallback shadows implemented ${name}")
 endif()
endforeach()
foreach(name Accessibility AVRouting)
 if(NOT TARGET load_only_${name})
  message(FATAL_ERROR "missing fallback ${name}")
 endif()
endforeach()
get_property(targets DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../../src/frameworks/load-only" PROPERTY BUILDSYSTEM_TARGETS)
list(LENGTH targets count)
if(NOT count EQUAL 58)
 message(FATAL_ERROR "expected58 fallbacks, got${count}")
endif()
